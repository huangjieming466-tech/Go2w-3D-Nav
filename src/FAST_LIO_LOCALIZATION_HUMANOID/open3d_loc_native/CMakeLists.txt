cmake_minimum_required(VERSION 3.8)
project(open3d_loc_native)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters registration)
add_executable(global_localization_node src/global_localization_node.cpp)
target_include_directories(global_localization_node PRIVATE ${PCL_INCLUDE_DIRS})
target_compile_definitions(global_localization_node PRIVATE ${PCL_DEFINITIONS})
target_link_libraries(global_localization_node ${PCL_LIBRARIES})
ament_target_dependencies(global_localization_node rclcpp geometry_msgs nav_msgs sensor_msgs std_msgs tf2 tf2_ros pcl_conversions)
install(TARGETS global_localization_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
